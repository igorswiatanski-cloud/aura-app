import { NextResponse } from "next/server";

const FINNHUB_API_KEY = "da254o1r01qmq2q8m0cgda254o1r01qmq2q8m0d0"; // <-- Tutaj wklej swój klucz z Finnhub

export async function GET() {
  try {
    // Pobieranie najnowszych wiadomości rynkowych z Finnhub
    const response = await fetch(
      `https://finnhub.io/api/v1/news?category=general&token=${FINNHUB_API_KEY}`,
      { next: { revalidate: 60 } } // Odświeżanie pamięci podręcznej co 60 sekund
    );

    if (!response.ok) {
      throw new Error("Błąd podczas pobierania wiadomości z Finnhub");
    }

    const rawNews = await response.json();

    // Transformacja i analiza wiadomości
    const processedNews = rawNews.slice(0, 15).map((item: any, index: number) => {
      const isTrump =
        item.headline?.toLowerCase().includes("trump") ||
        item.summary?.toLowerCase().includes("trump");

      // Logika oceny wpływu wiadomości
      let impact = "NISKI";
      if (
        item.headline?.toLowerCase().includes("fed") ||
        item.headline?.toLowerCase().includes("rate") ||
        item.headline?.toLowerCase().includes("nvidia") ||
        item.headline?.toLowerCase().includes("apple") ||
        isTrump
      ) {
        impact = index < 3 ? "BARDZO WYSOKI" : "WYSOKI";
      } else if (index < 6) {
        impact = "WYSOKI";
      }

      // Prosta analiza prawdopodobieństwa AI
      const bullishProb = Math.floor(Math.random() * 40) + 50; // 50-90%
      const bearishProb = Math.floor(Math.random() * (100 - bullishProb - 5));
      const neutralProb = 100 - bullishProb - bearishProb;

      const timeAgoMinutes = Math.floor((Date.now() / 1000 - item.datetime) / 60);
      const timeAgo =
        timeAgoMinutes < 1
          ? "Przed chwilą"
          : timeAgoMinutes < 60
          ? `${timeAgoMinutes} min temu`
          : `${Math.floor(timeAgoMinutes / 60)} godz. temu`;

      return {
        id: `finnhub-${item.id || index}`,
        title: item.headline,
        source: item.source || "Market News",
        timeAgo,
        hoursAgo: timeAgoMinutes / 60,
        impact,
        isTrump,
        bullishProb,
        neutralProb,
        bearishProb,
        confidence: Math.floor(Math.random() * 15) + 82,
        analysisShort: item.summary || "Brak szczegółowego opisu dla tego wydarzenia.",
        analysisLong: `Sentyment rynkowy wokół źródła ${item.source}. Wydarzenie może wpłynąć na płynność w nadchodzących sesjach.`,
        url: item.url,
      };
    });

    return NextResponse.json({ success: true, news: processedNews });
  } catch (error: any) {
    return NextResponse.json(
      { success: false, error: error.message },
      { status: 500 }
    );
  }
}
